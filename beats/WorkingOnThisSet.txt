const bass_one = ("~ 0 ~ ~ ~ 1 ~ ~")
const bass_two = ("~ 2 2 ~ ~ 0*2 ~ 1")
$: note(bass_one)
  .scale("c2:dorian")
  .sound("sawtooth")
  .lpf(650)
  .lpenv(28)
  .lpd(.15)
  .lpq("<4 8 12 16>")

  .shape(.4)
  .lastOf(4, x=>x.ply("2").dec(.1).add(note("[3 5 4 6]")))
