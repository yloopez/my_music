const allCC = await midin('MPK mini 3')
const cc = (ccNum) => allCC(ccNum, 1)
const kb = await midikeys('MPK mini 3')

_$: kb().s("supersaw").shape(.5)

const lpfilter_one = cc(16).range(100,2000)
const lpfilter_two = cc(17).range(100,2000)
const hpfilter = cc(20).range(0,1000)

all(x=>x.hpf(hpfilter))

const kick_one = "bd bd bd bd"
const kick_two = "[bd bd bd bd]*<1!6 2 [2 3]>"

_$kick: sound(kick_two)
  .bank("RolandTR909")
  .n("<1 2 2>")
  .distort("<0.8:.5!6 1:.4 1.5:.3>")
  .gain(.5)
  .off(-1/16, x=>x.speed(2).gain(.2))


const l_tom = s("~ lt:2 ~ lt:2 ~ ~ [lt:2 ~] lt:2")
const m_tom = s("mt:2 ~ [~ mt:2] ~ mt:2*2  ~ ~ ~")
const h_tom = s("~ ht:2*2 ~ ~ ~ [~ ht:2] ~ ht:2")

$toms: stack(
  //l_tom,
  //m_tom,
  //h_tom
).bank("RolandTR909")
  .gain(.6)
  .lpf(sine.range(200, 1000).slow(4))

const chords_one = "<Cm9 Am9>"
const chords_two = "<Cm9 Am9 Fm9 G7>"
_$sustain: chord(chords_two.slow(2))
  .dict('ireal')
  .sound("sine")
  .voicing()
  .lpf(320)
  .attack(0.2)
  .decay(3)
  .sustain(0)
  .release(1)
  .room(.4)
  .gain(.25)
  .vib("2:.3")
  .postgain(slider(0.045))

_$: note("<[<[4 7] [2 5] [0 3]> ~ ~ ~] [~ ~ ~ ~]>")
  .scale("c3:dorian")
  .sound("supersaw")
  .decay(.09)
  .vib("4:.3")
  .lpf(lpfilter_one)
  .lpq(10)
  .lpenv(3)
  .room(.3)
  .distort(".7")
  .gain(.4)

_$: n(irand(6).seg(16))
  .scale("c:dorian")
  .s("sawtooth")
  .lpf(lpfilter_two)
  .lpenv(4)
  .rib(46,1)
  .gain(.3)
  .mask("<1 0 0 0>*2")

_$hats: s("hh!8")
  .bank("RolandTR909")
  .room(.1)
  .sometimesBy(.2, x=>x.sound("oh | cp"))
  .postgain(slider(0.158))

_$notes: note("<[2 4]*2 [4 6]*2 [6 8]*2 [8 10]*2 [[~ 8] [7 ~] [1 2] 3]>*2")
  .scale("c3:dorian")
  .sound("supersaw")
  //.lpf(sine.range(200, 1000).slow(4))
  .lpf(lpfilter_one)
  .lpenv(4)
  .lpq(10)
  .room(.2)
  .postgain(slider(0.067))
  ._pianoroll()

_$bass: note("~ 2 ~ 1 ~ 0 1 ~")
  .scale("c2:dorian")
  .sound("square")
  .lpf(350)
  .lpenv(5)
  .lpq(8)
  .room(.4)
  .gain(.5)
